function countBinarySubstrings(s: string): number {
    var prev = 0;
    var cur = 0;
    var total = 0;
    for (var i = 0; i < s.length; i++) {
        if (i > 0 && s.charAt(i) !== s.charAt(i - 1)) {
            prev = cur;
            cur = 1;
        } else {
            cur++;
        }
        if (prev >= cur) total++;
    }
    return total;
}
