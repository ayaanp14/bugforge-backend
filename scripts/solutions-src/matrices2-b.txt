function diagonalSort(mat: number[][]): number[][] {
    var m = mat.length;
    var n = mat[0].length;
    var groups: { [key: string]: number[] } = {};
    for (var r = 0; r < m; r++) {
        for (var c = 0; c < n; c++) {
            var key = String(r - c);
            if (groups[key] === undefined) groups[key] = [];
            groups[key].push(mat[r][c]);
        }
    }
    for (var g in groups) {
        groups[g].sort(function (a, b) { return a - b; });
    }
    var cursor: { [key: string]: number } = {};
    var out: number[][] = [];
    for (var r2 = 0; r2 < m; r2++) {
        var row: number[] = [];
        for (var c2 = 0; c2 < n; c2++) {
            var key2 = String(r2 - c2);
            var idx = cursor[key2] === undefined ? 0 : cursor[key2];
            row.push(groups[key2][idx]);
            cursor[key2] = idx + 1;
        }
        out.push(row);
    }
    return out;
}

function minPathSum(grid: number[][]): number {
    var m = grid.length;
    var n = grid[0].length;
    var dp: number[][] = [];
    for (var i = 0; i < m; i++) dp.push(grid[i].slice());
    for (var c = 1; c < n; c++) dp[0][c] += dp[0][c - 1];

        }
    }
    return dp[m - 1][n - 1];
}

    var n = obstacleGrid[0].length;
    if (obstacleGrid[0][0] === 1) return 0;
    var dp: number[][] = [];
    for (var i = 0; i < m; i++) {
        var row: number[] = [];
        for (var j = 0; j < n; j++) row.push(0);
        dp.push(row);
    }
    dp[0][0] = 1;
    for (var r = 0; r < m; r++) {
        for (var c = 0; c < n; c++) {
            if (obstacleGrid[r][c] === 1) { dp[r][c] = 0; continue; }
            if (r > 0) dp[r][c] += dp[r - 1][c];
            if (c > 0) dp[r][c] += dp[r][c - 1];
        }
    }
    return dp[m - 1][n - 1];
}

            }
        }
    }
    return best * best;
}

function solve(board: number[][]): number[][] {
    var m = board.length;
    var n = board[0].length;
    var safe: boolean[][] = [];
    for (var i = 0; i < m; i++) {
        var srow: boolean[] = [];
        for (var j = 0; j < n; j++) srow.push(false);
        safe.push(srow);
    }
    var stack: number[][] = [];
    var push = function (r: number, c: number): void {
        if (r >= 0 && r < m && c >= 0 && c < n && board[r][c] === 1 && !safe[r][c]) {
            safe[r][c] = true;
            stack.push([r, c]);
        }
    };
    for (var r1 = 0; r1 < m; r1++) { push(r1, 0); push(r1, n - 1); }
    for (var c1 = 0; c1 < n; c1++) { push(0, c1); push(m - 1, c1); }
    while (stack.length > 0) {
        var cell = stack.pop() as number[];
        push(cell[0] - 1, cell[1]);
        push(cell[0] + 1, cell[1]);
        push(cell[0], cell[1] - 1);
        push(cell[0], cell[1] + 1);
    }
    var out: number[][] = [];
    for (var r2 = 0; r2 < m; r2++) {
        var row: number[] = [];
        for (var c2 = 0; c2 < n; c2++) row.push(safe[r2][c2] ? board[r2][c2] : 0);
        out.push(row);
    }
    return out;
}

    var n = mat.length;
    var cur: number[][] = [];

    var same = function (a: number[][], b: number[][]): boolean {
        for (var r = 0; r < n; r++) {
            for (var c = 0; c < n; c++) {
                if (a[r][c] !== b[r][c]) return false;
            }
        }
        return true;
    };
    for (var k = 0; k < 4; k++) {
        if (same(cur, target)) return true;
        var next: number[][] = [];
        for (var r2 = 0; r2 < n; r2++) {
            var row: number[] = [];
            for (var c2 = 0; c2 < n; c2++) row.push(cur[n - 1 - c2][r2]);
            next.push(row);
        }
        cur = next;
    }
    return false;
}
