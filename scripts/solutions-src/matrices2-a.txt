function generateMatrix(n: number): number[][] {
    var m: number[][] = [];
    for (var i = 0; i < n; i++) {
        var row: number[] = [];
        for (var j = 0; j < n; j++) row.push(0);
        m.push(row);
    }
    var top = 0, bottom = n - 1, left = 0, right = n - 1, v = 1;
    while (top <= bottom && left <= right) {
        for (var c = left; c <= right; c++) { m[top][c] = v; v++; }
        top++;
        for (var r = top; r <= bottom; r++) { m[r][right] = v; v++; }
        right--;
        if (top <= bottom) {
            for (var c2 = right; c2 >= left; c2--) { m[bottom][c2] = v; v++; }
            bottom--;
        }
        if (left <= right) {
            for (var r2 = bottom; r2 >= top; r2--) { m[r2][left] = v; v++; }
            left++;
        }
    }
    return m;
}

    var rows = mat.length;
    var cols = mat[0].length;
    var out: number[] = [];
    for (var d = 0; d < rows + cols - 1; d++) {
        if (d % 2 === 0) {
            var r = d < rows ? d : rows - 1;
            var c = d - r;
            while (r >= 0 && c < cols) {
                out.push(mat[r][c]);
                r--;
                c++;
            }
        } else {
            var c2 = d < cols ? d : cols - 1;
            var r2 = d - c2;
            while (c2 >= 0 && r2 < rows) {
                out.push(mat[r2][c2]);
                r2++;
                c2--;
            }
        }
    }
    return out;
}

function matrixReshape(mat: number[][], r: number, c: number): number[][] {
    var rows = mat.length;
    var cols = mat[0].length;
    if (rows * cols !== r * c) return mat;
    var flat: number[] = [];
    for (var i = 0; i < rows; i++) {
        for (var j = 0; j < cols; j++) flat.push(mat[i][j]);
    }
    var out: number[][] = [];
    for (var k = 0; k < r; k++) out.push(flat.slice(k * c, k * c + c));
    return out;
}

    int[][] out = new int[r][c];
    for (int k = 0; k < r * c; k++) {
        out[k / c][k % c] = mat[k / cols][k % cols];
    }
    return out;
}

    var n = grid[0].length;

        for (var c = 1; c < n; c++) {
            if (grid[r][c] > best) best = grid[r][c];
        }
        rowMax.push(best);
    }
    for (var c2 = 0; c2 < n; c2++) {
        var best2 = grid[0][c2];
        for (var r2 = 1; r2 < m; r2++) {
            if (grid[r2][c2] > best2) best2 = grid[r2][c2];
        }
        colMax.push(best2);

    var total = 0;
    for (var r3 = 0; r3 < m; r3++) {
        for (var c3 = 0; c3 < n; c3++) {
            total += Math.min(rowMax[r3], colMax[c3]) - grid[r3][c3];
        }
    }
    return total;
}

def maxIncreaseKeepingSkyline(grid)
  m = grid.length
  n = grid[0].length
  row_max = grid.map(&:max)
  col_max = (0...n).map { |c| (0...m).map { |r| grid[r][c] }.max }
  total = 0
  (0...m).each do |r|
    (0...n).each do |c|
      total += [row_max[r], col_max[c]].min - grid[r][c]
    end
  end
  total
