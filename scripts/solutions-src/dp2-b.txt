        var next: number[] = [];

            if (c + 1 < n && dp[c + 1] < best) best = dp[c + 1];

        if (dp[k] < answer) answer = dp[k];
    }
    return answer;
}

function stoneGame(piles: number[]): boolean {
    var n = piles.length;
    var dp: number[][] = [];
    for (var i = 0; i < n; i++) {
        var row: number[] = [];
        for (var j = 0; j < n; j++) row.push(0);
        dp.push(row);
    }
    for (var k = 0; k < n; k++) dp[k][k] = piles[k];
    for (var len = 2; len <= n; len++) {
        for (var a = 0; a + len - 1 < n; a++) {
            var b = a + len - 1;
            dp[a][b] = Math.max(piles[a] - dp[a + 1][b], piles[b] - dp[a][b - 1]);
        }
    }
    return dp[0][n - 1] > 0;
}

function lastStoneWeightII(stones: number[]): number {
    var total = 0;
    for (var i = 0; i < stones.length; i++) total += stones[i];
    var half = Math.floor(total / 2);
    var dp: boolean[] = [];
    for (var k = 0; k <= half; k++) dp.push(false);
    dp[0] = true;
    for (var s = 0; s < stones.length; s++) {
        for (var w = half; w >= stones[s]; w--) {
            if (dp[w - stones[s]]) dp[w] = true;
        }
    }
    for (var v = half; v >= 0; v--) {
        if (dp[v]) return total - 2 * v;
    }
    return total;
}

        var row: number[] = [];

fn findMaxForm(strs: Vec<String>, m: i32, n: i32) -> i32 {

    let cols = n as usize;

        let mut zeros = 0usize;
        let mut ones = 0usize;
        for b in s.as_bytes().iter() {
            if *b == b'0' {
                zeros += 1;
            } else {
                ones += 1;
            }
        }
        if zeros > rows || ones > cols {
            continue;
        }
        let mut i = rows;
        while i >= zeros {
            let mut j = cols;
            while j >= ones {
                let candidate = dp[i - zeros][j - ones] + 1;
                if candidate > dp[i][j] {
                    dp[i][j] = candidate;
                }
                if j == ones {
                    break;
                }
                j -= 1;
            }
            if i == zeros {
                break;
            }
            i -= 1;
        }
    }
    dp[rows][cols]
