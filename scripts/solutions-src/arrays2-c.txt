    var first: { [key: string]: number } = {};
    var last: { [key: string]: number } = {};
    var count: { [key: string]: number } = {};
    for (var i = 0; i < nums.length; i++) {
        var key = String(nums[i]);
        if (first[key] === undefined) first[key] = i;
        last[key] = i;
        count[key] = (count[key] === undefined ? 0 : count[key]) + 1;

    for (var k in count) {
        if (count[k] > degree) degree = count[k];

    var best = nums.length;
    for (var k2 in count) {
        if (count[k2] === degree) {
            var span = last[k2] - first[k2] + 1;
            if (span < best) best = span;
        }
    }
    return best;
}

function findLHS(nums: number[]): number {
    var count: { [key: string]: number } = {};
    for (var i = 0; i < nums.length; i++) {
        var key = String(nums[i]);
        count[key] = (count[key] === undefined ? 0 : count[key]) + 1;
    }
    var best = 0;
    for (var j = 0; j < nums.length; j++) {
        var higher = count[String(nums[j] + 1)];
        if (higher !== undefined) {
            var total = count[String(nums[j])] + higher;
            if (total > best) best = total;
        }
    }
    return best;
}

    for (int x : nums) count.put(x, count.getOrDefault(x, 0) + 1);
    List<Integer> out = new ArrayList<>();
    for (Map.Entry<Integer, Integer> e : count.entrySet()) {
        if (e.getValue() * 3 > nums.length) out.add(e.getKey());
    }
    Collections.sort(out);
    int[] res = new int[out.size()];
    for (int i = 0; i < res.length; i++) res[i] = out.get(i);
    return res;
}

int* majorityElement(int* nums, int numsSize, int* returnSize) {
    int* out = (int*) malloc(sizeof(int) * 4);
    int len = 0;
    for (int i = 0; i < numsSize; i++) {
        bool earlier = false;
        for (int j = 0; j < i; j++) {
            if (nums[j] == nums[i]) { earlier = true; break; }
        }
        if (earlier) continue;
        int c = 0;
        for (int j = 0; j < numsSize; j++) {
            if (nums[j] == nums[i]) c++;
        }
        if (c * 3 > numsSize) out[len++] = nums[i];
    }
    for (int i = 1; i < len; i++) {
        int key = out[i];
        int j = i - 1;
        while (j >= 0 && out[j] > key) { out[j + 1] = out[j]; j--; }
        out[j + 1] = key;
    }
    *returnSize = len;
    return out;
}

fn majorityElement(nums: Vec<i32>) -> Vec<i32> {
    use std::collections::HashMap;
    let mut count: HashMap<i32, i32> = HashMap::new();
    for x in nums.iter() {
        *count.entry(*x).or_insert(0) += 1;
    }
    let n = nums.len() as i32;
    let mut out: Vec<i32> = Vec::new();
    for (v, c) in count.iter() {
        if c * 3 > n {
            out.push(*v);
        }
    }
    out.sort();
    out
}
