function addBinary(a: string, b: string): string {
    var i = a.length - 1;
    var j = b.length - 1;
    var carry = 0;
    var out = "";
    while (i >= 0 || j >= 0 || carry > 0) {
        var total = carry;
        if (i >= 0) { total += a.charCodeAt(i) - 48; i--; }
        if (j >= 0) { total += b.charCodeAt(j) - 48; j--; }
        out = String(total % 2) + out;
        carry = total > 1 ? 1 : 0;
    }
    return out;
}

  fact = [1, 1, 2, 6, 24, 120, 720, 5040, 40320, 362880]
  total = 0
  x = n
  while x > 0
    total += fact[x % 10]
    x /= 10
  end
  total == n
end

function isAutomorphic(n: number): boolean {
    var power = 1;
    var x = n;
    while (x > 0) {
        power *= 10;
        x = Math.floor(x / 10);
    }
    return (n * n) % power === n;
}
