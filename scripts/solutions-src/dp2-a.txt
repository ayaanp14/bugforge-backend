function deleteAndEarn(nums: number[]): number {
    var maxV = 0;
    for (var i = 0; i < nums.length; i++) {
        if (nums[i] > maxV) maxV = nums[i];
    }
    var points: number[] = [];
    for (var k = 0; k <= maxV; k++) points.push(0);
    for (var j = 0; j < nums.length; j++) points[nums[j]] += nums[j];
    var take = 0;
    var skip = 0;
    for (var v = 1; v <= maxV; v++) {
        var newTake = skip + points[v];
        var newSkip = Math.max(skip, take);
        take = newTake;
        skip = newSkip;
    }
    return Math.max(take, skip);
}

function findLength(nums1: number[], nums2: number[]): number {
    var m = nums1.length;
    var n = nums2.length;
    var dp: number[][] = [];
    for (var i = 0; i <= m; i++) {
        var row: number[] = [];
        for (var j = 0; j <= n; j++) row.push(0);
        dp.push(row);
    }
    var best = 0;
    for (var a = 1; a <= m; a++) {
        for (var b = 1; b <= n; b++) {
            if (nums1[a - 1] === nums2[b - 1]) {
                dp[a][b] = dp[a - 1][b - 1] + 1;
                if (dp[a][b] > best) best = dp[a][b];
            }
        }
    }
    return best;
}

    var best = 1;

                    count[i] += count[j];

    }

        if (len[m] === best) total += count[m];
    }
    return total;

function minDistance(word1: string, word2: string): number {
    var m = word1.length;
    var n = word2.length;
    var dp: number[][] = [];
    for (var i = 0; i <= m; i++) {
        var row: number[] = [];
        for (var j = 0; j <= n; j++) row.push(0);
        dp.push(row);
    }
    for (var a = 1; a <= m; a++) {
        for (var b = 1; b <= n; b++) {
            if (word1.charAt(a - 1) === word2.charAt(b - 1)) dp[a][b] = dp[a - 1][b - 1] + 1;
            else dp[a][b] = Math.max(dp[a - 1][b], dp[a][b - 1]);
        }
    }
    return m + n - 2 * dp[m][n];
}
