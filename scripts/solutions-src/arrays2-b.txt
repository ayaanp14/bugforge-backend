function thirdMax(nums: number[]): number {
    var seen: { [key: string]: boolean } = {};
    var distinct: number[] = [];
    for (var i = 0; i < nums.length; i++) {
        var key = String(nums[i]);
        if (seen[key] !== true) {
            seen[key] = true;
            distinct.push(nums[i]);

}
