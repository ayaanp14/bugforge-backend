function reorderSpaces(text: string): string {
    var words: string[] = [];
    var pieces = text.split(" ");
    for (var i = 0; i < pieces.length; i++) {
        if (pieces[i].length > 0) words.push(pieces[i]);
    }
    var spaces = 0;
    for (var j = 0; j < text.length; j++) {
        if (text.charAt(j) === " ") spaces++;
    }
    var pad = function (n: number): string {
        var s = "";
        for (var k = 0; k < n; k++) s += " ";
        return s;
    };
    if (words.length === 1) return words[0] + pad(spaces);
    var between = Math.floor(spaces / (words.length - 1));
    var extra = spaces - between * (words.length - 1);
    return words.join(pad(between)) + pad(extra);
}

        if (text[i] == ' ') spaces++;
    }

        if (w > 0) {
