    var lo = nums[0];
    var hi = nums[0];
    for (var i = 1; i < nums.length; i++) {
        if (nums[i] < lo) lo = nums[i];

function selfDividingNumbers(left: number, right: number): number[] {
    var out: number[] = [];
    for (var n = left; n <= right; n++) {
        var x = n;
        var ok = true;
        while (x > 0) {
            var d = x % 10;
            if (d === 0 || n % d !== 0) { ok = false; break; }
            x = Math.floor(x / 10);
        }
        if (ok) out.push(n);
    }
    return out;
}
