function findMinHeightTrees(n: number, edges: number[][]): number[] {
    if (n === 1) return [0];
    var adjacency: number[][] = [];
    var degree: number[] = [];
    for (var i = 0; i < n; i++) {
        adjacency.push([]);
        degree.push(0);
    }
    for (var j = 0; j < edges.length; j++) {
        adjacency[edges[j][0]].push(edges[j][1]);
        adjacency[edges[j][1]].push(edges[j][0]);
        degree[edges[j][0]]++;
        degree[edges[j][1]]++;
    }
    var leaves: number[] = [];
    for (var k = 0; k < n; k++) {
        if (degree[k] === 1) leaves.push(k);
    }
    var remaining = n;
    while (remaining > 2) {
        remaining -= leaves.length;
        var next: number[] = [];
        for (var m = 0; m < leaves.length; m++) {
            var leaf = leaves[m];
            for (var p = 0; p < adjacency[leaf].length; p++) {
                var neighbour = adjacency[leaf][p];
                degree[neighbour]--;
                if (degree[neighbour] === 1) next.push(neighbour);
            }
        }
        leaves = next;
    }
    return leaves.slice().sort(function (a, b) { return a - b; });
}

        var seen: boolean[][] = [];

            var row: boolean[] = [];
            for (var c = 0; c < cols; c++) row.push(false);
            seen.push(row);

        while (stack.length > 0) {
            var cell = stack.pop() as number[];
            if (cell[0] === rows - 1 && cell[1] === cols - 1) return true;
            for (var d = 0; d < 4; d++) {
                var nr = cell[0] + dr[d];
                var nc = cell[1] + dc[d];
                if (nr < 0 || nc < 0 || nr >= rows || nc >= cols || seen[nr][nc]) continue;
                var diff = Math.abs(heights[nr][nc] - heights[cell[0]][cell[1]]);
                if (diff > limit) continue;
                seen[nr][nc] = true;
                stack.push([nr, nc]);
            }

        return seen[rows - 1][cols - 1];

        for (var c2 = 0; c2 < cols; c2++) {
            if (heights[r2][c2] > hi) hi = heights[r2][c2];

        var mid = Math.floor((lo + hi) / 2);
        if (reachable(mid)) hi = mid;
        else lo = mid + 1;
