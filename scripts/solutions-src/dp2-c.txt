    var isPredecessor = function (a: string, b: string): boolean {
        if (b.length !== a.length + 1) return false;
        var i = 0;
        var j = 0;
        var skips = 0;
        while (i < a.length && j < b.length) {
            if (a.charAt(i) === b.charAt(j)) { i++; j++; }
            else {
                skips++;
                j++;
                if (skips > 1) return false;
            }
        }
        return true;
    };
    var sorted = words.slice();
    sorted.sort(function (a, b) { return a.length - b.length; });
    var best: number[] = [];
    for (var k = 0; k < sorted.length; k++) best.push(1);
    var answer = 1;

        for (var y = 0; y < x; y++) {
            if (isPredecessor(sorted[y], sorted[x]) && best[y] + 1 > best[x]) best[x] = best[y] + 1;
        }
        if (best[x] > answer) answer = best[x];

            var thirty = dp[Math.max(0, d - 30)] + costs[2];
            dp[d] = Math.min(one, Math.min(seven, thirty));
        }
    }
    return dp[last];
}

        array(4, 6), array(6, 8), array(7, 9), array(4, 8), array(3, 9, 0),

    for ($step = 1; $step < $n; $step++) {
        $next = array_fill(0, 10, 0);
        for ($digit = 0; $digit < 10; $digit++) {
            foreach ($moves[$digit] as $to) {
                $next[$digit] = ($next[$digit] + $dp[$to]) % $MOD;
            }
        }
        $dp = $next;
    }
    $total = 0;
    for ($d = 0; $d < 10; $d++) $total = ($total + $dp[$d]) % $MOD;

}

def knightDialer(n)
  mod = 1000000007
  moves = [[4, 6], [6, 8], [7, 9], [4, 8], [3, 9, 0], [], [1, 7, 0], [2, 6], [1, 3], [2, 4]]
  dp = Array.new(10, 1)
  (1...n).each do
    nxt = Array.new(10, 0)
    (0...10).each do |digit|
      moves[digit].each do |to|
        nxt[digit] = (nxt[digit] + dp[to]) % mod
      end
    end
    dp = nxt
  end
  dp.sum % mod
end
