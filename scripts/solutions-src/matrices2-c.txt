        }
    }

    var rows: boolean[][] = [];
    var cols: boolean[][] = [];
    var boxes: boolean[][] = [];
    for (var i = 0; i < 9; i++) {
        var a: boolean[] = [];
        var b: boolean[] = [];

        rows.push(a);
        cols.push(b);
        boxes.push(d);
    }
    for (var r = 0; r < 9; r++) {
        for (var c = 0; c < 9; c++) {
            var v = board[r][c];
            if (v === 0) continue;
            var box = Math.floor(r / 3) * 3 + Math.floor(c / 3);

            rows[r][v] = true;
            cols[c][v] = true;
            boxes[box][v] = true;
        }
    }
    return true;
}

    vector<int> rowSum(m, 0), colSum(n, 0);
    for (int r = 0; r < m; r++) {
        for (int c = 0; c < n; c++) {
            rowSum[r] += mat[r][c];
            colSum[c] += mat[r][c];
        }
    }
    int count = 0;
    for (int r = 0; r < m; r++) {
        for (int c = 0; c < n; c++) {
            if (mat[r][c] == 1 && rowSum[r] == 1 && colSum[c] == 1) count++;
        }
    }
    return count;
}
