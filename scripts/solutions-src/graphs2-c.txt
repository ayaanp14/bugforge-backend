function countPairs(n: number, edges: number[][]): number {
    var parent: number[] = [];
    var size: number[] = [];
    for (var i = 0; i < n; i++) {
        parent.push(i);
        size.push(1);
    }
    var find = function (x: number): number {
        while (parent[x] !== x) {
            parent[x] = parent[parent[x]];
            x = parent[x];
        }
        return x;
    };
    for (var j = 0; j < edges.length; j++) {
        var a = find(edges[j][0]);
        var b = find(edges[j][1]);
        if (a !== b) {
            parent[a] = b;
            size[b] += size[a];
        }
    }
    var unreachable = 0;
    var seenSoFar = 0;
    for (var k = 0; k < n; k++) {
        if (find(k) !== k) continue;
        unreachable += size[k] * seenSoFar;
        seenSoFar += size[k];
    }
    return unreachable;
}

function snakesAndLadders(board: number[][]): number {
    var n = board.length;
    var target = n * n;
    var valueAt = function (square: number): number {
        var quotient = Math.floor((square - 1) / n);
        var remainder = (square - 1) % n;
        var row = n - 1 - quotient;
        var col = quotient % 2 === 0 ? remainder : n - 1 - remainder;
        return board[row][col];
    };
    var dist: number[] = [];
    for (var i = 0; i <= target; i++) dist.push(-1);
    dist[1] = 0;
    var queue: number[] = [1];
    var head = 0;
    while (head < queue.length) {
        var square = queue[head++];
        if (square === target) return dist[square];
        for (var step = 1; step <= 6 && square + step <= target; step++) {
            var landing = square + step;
            var jump = valueAt(landing);
            var destination = jump === -1 ? landing : jump;
            if (dist[destination] === -1) {
                dist[destination] = dist[square] + 1;
                queue.push(destination);
            }
        }
    }
    return -1;
}

            }

                stack[top++] = index;
            }
        }

    while (top > 0) {

function swimInWater(grid: number[][]): number {

    var dr = [1, -1, 0, 0];
    var dc = [0, 0, 1, -1];
    var reachable = function (limit: number): boolean {
        if (grid[0][0] > limit) return false;
        var seen: boolean[][] = [];
        for (var r = 0; r < n; r++) {
            var row: boolean[] = [];
            for (var c = 0; c < n; c++) row.push(false);
            seen.push(row);
        }
        seen[0][0] = true;
        var stack: number[][] = [[0, 0]];
        while (stack.length > 0) {
            var cell = stack.pop() as number[];
            if (cell[0] === n - 1 && cell[1] === n - 1) return true;
            for (var d = 0; d < 4; d++) {
                var nr = cell[0] + dr[d];
                var nc = cell[1] + dc[d];
                if (nr < 0 || nc < 0 || nr >= n || nc >= n || seen[nr][nc]) continue;
                if (grid[nr][nc] > limit) continue;
                seen[nr][nc] = true;
                stack.push([nr, nc]);
            }
        }
        return seen[n - 1][n - 1];

    var lo = grid[0][0] > grid[n - 1][n - 1] ? grid[0][0] : grid[n - 1][n - 1];

    while (lo < hi) {
        var mid = Math.floor((lo + hi) / 2);

    let mut seen = vec![vec![false; n]; n];
    seen[0][0] = true;
    let mut stack: Vec<(usize, usize)> = vec![(0, 0)];
    let dr: [i32; 4] = [1, -1, 0, 0];
    let dc: [i32; 4] = [0, 0, 1, -1];
    while let Some((cr, cc)) = stack.pop() {
        if cr == n - 1 && cc == n - 1 {
            return true;
        }
        for d in 0..4 {
            let nr = cr as i32 + dr[d];
            let nc = cc as i32 + dc[d];
            if nr < 0 || nc < 0 || nr >= n as i32 || nc >= n as i32 {
                continue;
            }
            let (nru, ncu) = (nr as usize, nc as usize);
            if seen[nru][ncu] || grid[nru][ncu] > limit {
                continue;
            }
            seen[nru][ncu] = true;
            stack.push((nru, ncu));
        }
    }
    seen[n - 1][n - 1]
}

fn swimInWater(grid: Vec<Vec<i32>>) -> i32 {
    let n = grid.len();
    let mut lo = if grid[0][0] > grid[n - 1][n - 1] { grid[0][0] } else { grid[n - 1][n - 1] };
    let mut hi = (n * n - 1) as i32;
    while lo < hi {
        let mid = lo + (hi - lo) / 2;
        if swim_reachable(&grid, mid) {
            hi = mid;
        } else {
            lo = mid + 1;
        }
    }
    lo
}
