function maximumSwap(num: number): number {
    var digits = String(num).split("");
    var lastIndex: number[] = [];
    for (var i = 0; i < 10; i++) lastIndex.push(-1);
    for (var j = 0; j < digits.length; j++) lastIndex[Number(digits[j])] = j;
    for (var k = 0; k < digits.length; k++) {
        for (var d = 9; d > Number(digits[k]); d--) {
            if (lastIndex[d] > k) {
                var idx = lastIndex[d];
                var t = digits[k];
                digits[k] = digits[idx];
                digits[idx] = t;
                return parseInt(digits.join(""), 10);
            }
        }
    }
    return num;
}
