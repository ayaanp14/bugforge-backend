function topKFrequent(words: string[], k: number): string[] {
    var count: { [key: string]: number } = {};
    for (var i = 0; i < words.length; i++) {
        count[words[i]] = (count[words[i]] || 0) + 1;
    }
    var unique: string[] = [];
    for (var w in count) {
        if (Object.prototype.hasOwnProperty.call(count, w)) unique.push(w);
    }
    unique.sort(function (a, b) {
        if (count[a] !== count[b]) return count[b] - count[a];
        if (a < b) return -1;
        if (a > b) return 1;
        return 0;
    });
    return unique.slice(0, k);
}
