    var count: { [key: string]: number } = {};
    for (var i = 0; i < s.length; i++) {
        var ch = s.charAt(i);
        count[ch] = (count[ch] === undefined ? 0 : count[ch]) + 1;
    }
    var total = 0;
    var hasOdd = false;
    for (var k in count) {
        total += count[k] - (count[k] % 2);
        if (count[k] % 2 === 1) hasOdd = true;
    }
    return total + (hasOdd ? 1 : 0);
}
