function reverse(x: number): number {
    var sign = x < 0 ? -1 : 1;
    var n = Math.abs(x);
    var result = 0;
    while (n > 0) {
        result = result * 10 + (n % 10);
        n = Math.floor(n / 10);
    }
    result *= sign;
    if (result < -2147483648 || result > 2147483647) return 0;
    return result;
}

function countPrimes(n: number): number {
    if (n < 3) return 0;
    var sieve: boolean[] = [];
    for (var k = 0; k < n; k++) sieve.push(false);
    var count = 0;
    for (var i = 2; i < n; i++) {
        if (!sieve[i]) {
            count++;
            for (var j = i * i; j < n; j += i) sieve[j] = true;
        }
    }
    return count;
}
