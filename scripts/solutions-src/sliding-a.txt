function findMaxLength(nums: number[]): number {
    var firstAt: { [key: string]: number } = {};
    firstAt["0"] = -1;
    var balance = 0;
    var best = 0;
    for (var i = 0; i < nums.length; i++) {
        balance += nums[i] === 1 ? 1 : -1;
        var key = String(balance);
        if (firstAt[key] !== undefined) {

            if (len > best) best = len;
        } else {
            firstAt[key] = i;
        }
    }
    return best;
}
