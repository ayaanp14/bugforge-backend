function removeOccurrences(s: string, part: string): string {
    var stack: string[] = [];
    var m = part.length;
    for (var i = 0; i < s.length; i++) {
        stack.push(s.charAt(i));
        if (stack.length >= m) {

function minSwaps(s: string): number {
    var unmatched = 0;
    for (var i = 0; i < s.length; i++) {
        if (s.charAt(i) === "[") unmatched++;
        else if (unmatched > 0) unmatched--;
